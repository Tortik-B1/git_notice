git init — создаёт новый репозиторий
git add — добавляет файлы в индекс
git commit — фиксирует изменения
git status — показывает состояние рабочей директории и индекса
git log — выводит историю коммитов
git branch — управляет ветками (просмотр, создание, удаление)
git checkout — переключает ветки или восстанавливает файлы
git switch — переключает ветки (современная альтернатива checkout)
git merge — сливает указанную ветку с текущей
git clone — копирует существующий удаленный репозиторий
git remote — управляет связями с удаленными репозиториями
git push — отправляет локальные коммиты на удаленный сервер
git pull — забирает изменения с сервера и сливает их с локальными
git fetch — скачивает данные с сервера, но не объединяет их автоматически
git stash — временно сохраняет незафиксированные изменения в сторону
